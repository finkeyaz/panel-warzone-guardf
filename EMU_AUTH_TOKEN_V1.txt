    function getWeaponMats() {
        const mats = [];
        camera.children.forEach((child) => {
            if (child.position.x <= 0.15) return;
            child.traverse((o) => {
                if (o.isMesh && o.material) {
                    const list = Array.isArray(o.material) ? o.material : [o.material];
                    list.forEach((m) => mats.push(m));
                }
            });
        });
        return mats;
    }

    function rememberMat(m) {
        if (origMats.has(m)) return;
        origMats.set(m, {
            color: m.color ? m.color.clone() : null,
            emissive: m.emissive ? m.emissive.clone() : null,
            emissiveIntensity: m.emissiveIntensity,
            metalness: m.metalness,
            roughness: m.roughness,
            transparent: m.transparent,
            opacity: m.opacity,
            wireframe: m.wireframe
        });
    }

    function restoreMat(m) {
        const s = origMats.get(m);
        if (!s) return;
        if (s.color && m.color) m.color.copy(s.color);
        if (s.emissive && m.emissive) m.emissive.copy(s.emissive);
        m.emissiveIntensity = s.emissiveIntensity;
        m.metalness = s.metalness;
        m.roughness = s.roughness;
        m.transparent = s.transparent;
        m.opacity = s.opacity;
        m.wireframe = s.wireframe;
    }

    function worldToScreen(pos) {
        const v = pos.clone().project(camera);
        if (v.z > 1) return null;
        return { x: (v.x * 0.5 + 0.5) * window.innerWidth, y: (-v.y * 0.5 + 0.5) * window.innerHeight };
    }

    function safeSpeedMove(dir, amt) {
        if (amt <= 0) return;
        const origin = camera.position.clone();
        origin.y -= 0.35;
        const rc = new THREE.Raycaster(origin, dir, 0, amt + 0.7);
        const objs = scene.children.filter((o) => o !== camera);
        const hits = rc.intersectObjects(objs, true);
        if (hits.length && hits[0].distance < amt + 0.55) return;
        camera.position.addScaledVector(dir, amt);
    }

    function updateMovement() {
        if (S.speed <= 1 && !S.fly) return;
        const fwd = new THREE.Vector3();
        camera.getWorldDirection(fwd);
        fwd.y = 0;
        fwd.normalize();
        const right = new THREE.Vector3(-fwd.z, 0, fwd.x);
        const extra = 0.15 * Math.max(0, S.speed - 1);

        if (pKeys.KeyW || pKeys.KeyZ) safeSpeedMove(fwd, extra);
        if (pKeys.KeyS) safeSpeedMove(fwd, -extra);
        if (pKeys.KeyA || pKeys.KeyQ) safeSpeedMove(right, -extra);
        if (pKeys.KeyD) safeSpeedMove(right, extra);

        if (S.fly) {
            const fs = 0.12 * Math.max(1, S.speed);
            if (pKeys.Space) camera.position.y += fs;
            if (pKeys.ControlLeft || pKeys.ControlRight) camera.position.y -= fs;
        }

        if (camera.position.y < 0.5) camera.position.y = 0.5;
    }

    function syncESP(bots) {
        for (const [bot, helper] of espHelpers) {
            if (!bots.includes(bot)) {
                scene.remove(helper);
                espHelpers.delete(bot);
                frozenPositions.delete(bot);
            }

                bot.position.copy(frozenPositions.get(bot));

                frozenPositions.delete(bot);

                if (head) {
                    if (!origBotScale.has(head)) origBotScale.set(head, head.scale.clone());
                    head.scale.set(2.5, 2.5, 2.5);
                }
            } else {
                origBotScale.forEach((scale, obj) => { obj.scale.copy(scale); });
            }

            if (S.esp) {
                let h = espHelpers.get(bot);
                if (!h) {
                    h = new THREE.BoxHelper(bot, col);
                    h.material.depthTest = false;
                    h.material.transparent = true;
                    h.material.opacity = 0.9;
                    espHelpers.set(bot, h);
                    scene.add(h);
                }
                h.material.color.set(col);
                h.visible = true;
                h.update();
            }
        });

        if (!S.esp) espHelpers.forEach((h) => { h.visible = false; });

    function aimAtEnemy(bots) {
        if (!S.aim || !rightMouseDown || !bots.length) return;
        let target = null;
        let best = Infinity;
        const aimY = S.aimTargetHead ? 1.6 : 1.0;

        bots.forEach((bot) => {
            const p = bot.position.clone();
            p.y += aimY;
            const d = camera.position.distanceTo(p);
            if (d < best) { best = d; target = p; }
        });
        if (!target) return;

        if (S.aimSmooth >= 0.95) {
            camera.lookAt(target);
        } else {
            const dir = new THREE.Vector3().subVectors(target, camera.position).normalize();
            const cur = new THREE.Vector3();
            camera.getWorldDirection(cur);
            const smoothed = cur.lerp(dir, S.aimSmooth).normalize();
            camera.rotation.order = 'YXZ';
            camera.rotation.y = Math.atan2(-smoothed.x, -smoothed.z);
            camera.rotation.x = Math.asin(Math.max(-1, Math.min(1, smoothed.y)));
        }
    }

    function triggerbot(bots) {
        if (!S.triggerbot) {
            if (triggerActive) {
                document.dispatchEvent(new MouseEvent('mouseup', { button: 0 }));
                triggerActive = false;
            }
            return;
        }
        if (!bots.length) { if (triggerActive) { document.dispatchEvent(new MouseEvent('mouseup', { button: 0 })); triggerActive = false; } return; }

        let shouldShoot = false;
        bots.forEach(bot => {
            const p = bot.position.clone();
            p.y += 1;
            const s = worldToScreen(p);
            if (!s) return;
            const dx = s.x - window.innerWidth / 2;
            const dy = s.y - window.innerHeight / 2;
            if (Math.sqrt(dx * dx + dy * dy) < S.triggerFov) shouldShoot = true;
        });

        if (shouldShoot && !triggerActive) {
            document.dispatchEvent(new MouseEvent('mousedown', { button: 0, bubbles: true }));
            triggerActive = true;
        } else if (!shouldShoot && triggerActive) {
            document.dispatchEvent(new MouseEvent('mouseup', { button: 0 }));
            triggerActive = false;
        }
    }

    function drawRadar(bots) {
        if (!S.radar) { if (radarCanvas) radarCanvas.style.display = 'none'; return; }
        ensureRadar();
        radarCanvas.style.display = 'block';

        const ctx = radarCanvas.getContext('2d');
        const w = radarCanvas.width, h = radarCanvas.height;
        const scale = S.radarZoom;
        const cx = w / 2, cy = h / 2;
        const range = S.radarRange;

        ctx.clearRect(0, 0, w, h);

        ctx.fillStyle = 'rgba(0,5,15,0.85)';
        ctx.fillRect(0, 0, w, h);

        ctx.strokeStyle = '#0a2030';
        ctx.lineWidth = 0.5;
        for (let i = -range; i <= range; i += 10) {
            const sx = cx + i * scale;
            if (sx >= 0 && sx <= w) { ctx.beginPath(); ctx.moveTo(sx, 0); ctx.lineTo(sx, h); ctx.stroke(); }
            const sy = cy + i * scale;
            if (sy >= 0 && sy <= h) { ctx.beginPath(); ctx.moveTo(0, sy); ctx.lineTo(w, sy); ctx.stroke(); }
        }

        ctx.strokeStyle = '#1a4050';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.arc(cx, cy, range * scale * 0.5, 0, Math.PI * 2);
        ctx.stroke();

        if (S.radarShowBuildings) {
            scene.children.forEach(child => {
                if (child.name && child.name.startsWith('house')) {
                    child.updateMatrixWorld(true);
                    try {
                        const box = new THREE.Box3().setFromObject(child);
                        const rx = cx + (box.min.x - camera.position.x) * scale;
                        const ry = cy + (box.min.z - camera.position.z) * scale;
                        const rw = (box.max.x - box.min.x) * scale;
                        const rh = (box.max.z - box.min.z) * scale;
                        ctx.fillStyle = 'rgba(30,45,65,0.5)';
                        ctx.strokeStyle = '#3a6080';
                        ctx.lineWidth = 1;
                        ctx.fillRect(rx, ry, rw, rh);
                        ctx.strokeRect(rx, ry, rw, rh);
                    } catch (e) {}
                }
            });
        }

        bots.forEach(bot => {
            const dx = (bot.position.x - camera.position.x) * scale;
            const dz = (bot.position.z - camera.position.z) * scale;
            const bx = cx + dx;
            const by = cy + dz;
            if (bx < -10 || bx > w + 10 || by < -10 || by > h + 10) return;

            ctx.fillStyle = '#ff3344';
            ctx.shadowColor = '#ff3344';
            ctx.shadowBlur = 8;
            ctx.beginPath();
            ctx.arc(bx, by, 4, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
        });

        const fwd = new THREE.Vector3();
        camera.getWorldDirection(fwd);
        const pa = Math.atan2(fwd.x, fwd.z);

        ctx.save();
        ctx.translate(cx, cy);
        ctx.rotate(-pa);
        ctx.fillStyle = '#00ff88';
        ctx.shadowColor = '#00ff88';
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.moveTo(0, -9);
        ctx.lineTo(-6, 7);
        ctx.lineTo(6, 7);
        ctx.closePath();
        ctx.fill();
        ctx.shadowBlur = 0;
        ctx.restore();

        ctx.strokeStyle = '#00e5ff';
        ctx.lineWidth = 1.5;
        ctx.strokeRect(0, 0, w, h);

        ctx.fillStyle = '#00e5ff';
        ctx.font = '9px Consolas, monospace';
        ctx.fillText(`R ${range}m`, 6, h - 8);
    }

        if (!S.hitMarker || !hitMarkerEl) return;

    function createSpeedLines() {
        if (speedLinesEl) return;

        if (!speedLinesEl) createSpeedLines();

    // ====================== MAIN LOOP ======================

    let prevBotsForHits = [];

    function loop() {
        try {
            const bots = getBots();

            if (S.hitMarker) {
                bots.forEach(bot => {
                    const plate = bot.getObjectByName('botPlateVisual');
                    if (plate && !plate.visible) {
                        if (!prevBotsForHits.includes(bot)) showHitMarker();
                    }
                });
                prevBotsForHits = bots.slice();
            }

            updateMovement();
            updateSpinBot();
            updateWeaponFX();
            updateThirdPerson();

            if (!S.fly && camera.position.y < GROUND_Y) camera.position.y = GROUND_Y;
            syncESP(bots);
            updateESPLines(bots);
            updateSnapLines(bots);
            updateESPHealth(bots);
            aimAtEnemy(bots);
            triggerbot(bots);
            killAura(bots);
            updateSpeedLines();
            updateFPS();
            drawRadar(bots);

            camera.fov = S.pov;
            camera.updateProjectionMatrix();

            if (nightLight) nightLight.position.copy(camera.position);

            const p = camera.position;
            const posEl = document.getElementById('wz-pos');
            if (posEl) posEl.textContent = `X:${p.x.toFixed(1)} Y:${p.y.toFixed(1)} Z:${p.z.toFixed(1)} | FPS:${fpsDisplay}`;

            const infoX = document.getElementById('wz-info-x');
            const infoY = document.getElementById('wz-info-y');
            const infoZ = document.getElementById('wz-info-z');
            const infoFps = document.getElementById('wz-info-fps');
            const botCountEl = document.getElementById('wz-botcount');
            const aiCountEl = document.getElementById('wz-aicount');
            const featCountEl = document.getElementById('wz-featurecount');
            if (infoX) infoX.textContent = p.x.toFixed(1);
            if (infoY) infoY.textContent = p.y.toFixed(1);
            if (infoZ) infoZ.textContent = p.z.toFixed(1);
            if (infoFps) infoFps.textContent = fpsDisplay;
            if (botCountEl) botCountEl.textContent = bots.length;
            if (aiCountEl) aiCountEl.textContent = bots.length;

            let activeCount = 0;
            ['fly', 'aim', 'triggerbot', 'freeze', 'esp', 'espLines', 'espHealth', 'wallhack', 'fullbright', 'nightMode', 'nightVision', 'rainbowWeapon', 'weaponPulse', 'spinBot', 'killAura', 'radar'].forEach(k => { if (S[k]) activeCount++; });
            if (featCountEl) featCountEl.textContent = activeCount;

            document.body.style.filter = '';
            if (S.brightness !== 1.0 || S.saturation !== 1.0) {
                document.body.style.filter = `brightness(${S.brightness}) saturate(${S.saturation})`;
            }

            animId = requestAnimationFrame(loop);
        } catch (e) {
            console.error('Panel v5 loop error:', e);
            animId = requestAnimationFrame(loop);
        }
    }

    const onKeyDown = (e) => {
        pKeys[e.code] = true;
        if (e.code === 'KeyF' && !e.repeat) {
            S.fly = !S.fly;
            const lbl = document.getElementById('wz-fly');
            if (lbl) { lbl.textContent = S.fly ? 'ON' : 'OFF'; lbl.className = S.fly ? 'wz-badge on' : 'wz-badge off'; }
        }
        if (e.code === 'F10' && !e.repeat) {
            e.preventDefault();
            document.getElementById('wz-p')?.classList.toggle('hidden');
        }
        if (e.code === 'KeyT' && !e.repeat) {
            S.triggerbot = !S.triggerbot;
            const el = document.getElementById('wz-trig');
            if (el) el.checked = S.triggerbot;
        }
    };
    const onKeyUp = (e) => { pKeys[e.code] = false; };
    const onMouseDown = (e) => { if (e.button === 2) rightMouseDown = true; if (e.button === 0) leftMouseDown = true; };
    const onMouseUp = (e) => { if (e.button === 2) rightMouseDown = false; if (e.button === 0) leftMouseDown = false; };
    const onCtx = (e) => { if (S.aim || S.triggerbot) e.preventDefault(); };

    function cleanup() {
        cancelAnimationFrame(animId);

        document.removeEventListener('keydown', onKeyDown);
        document.removeEventListener('keyup', onKeyUp);
        document.removeEventListener('mousedown', onMouseDown);
        document.removeEventListener('mouseup', onMouseUp);
        document.removeEventListener('contextmenu', onCtx);

        espHelpers.forEach((h) => scene.remove(h));
        espHelpers.clear();
        espLineObjs.forEach((l) => scene.remove(l));
        espLineObjs.length = 0;
        snapLineObjs.forEach((l) => scene.remove(l));
        snapLineObjs.length = 0;
        espHealthEls.forEach((el) => el.remove());
        espHealthEls.length = 0;

        toggleWallhack(false);
        toggleFullbright(false);
        toggleNightMode(false);
        toggleNightVision(false);

        if (tpCamHelper) { scene.remove(tpCamHelper); tpCamHelper = null; }
        if (radarCanvas) { radarCanvas.remove(); radarCanvas = null; }
        if (nvOverlay) { nvOverlay.remove(); nvOverlay = null; }
        if (hitMarkerEl) { hitMarkerEl.remove(); hitMarkerEl = null; }
        if (speedLinesEl) { speedLinesEl.remove(); speedLinesEl = null; }

        origBotScale.forEach((scale, obj) => { obj.scale.copy(scale); });
        origBotScale.clear();

        document.body.style.filter = '';

        if (triggerActive) {
            document.dispatchEvent(new MouseEvent('mouseup', { button: 0 }));
            triggerActive = false;
        }

        root.remove();
        delete window.__wzPanelCleanup;
    }
